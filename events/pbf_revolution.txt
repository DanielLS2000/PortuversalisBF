namespace = pbf_revolution

# First Coalition
country_event = {
    id = pbf_revolution.1
	title = pbf_revolution.1.t
	desc = pbf_revolution.1.d
	picture = REVOLUTION_eventPicture
	
	major = yes
    fire_only_once = yes
	
	trigger = {
		has_dlc = "Emperor"
		is_revolution_target = yes
		is_bankrupt = no
        is_at_war = no
        NOT = { has_global_flag = first_coalition }
	}

	mean_time_to_happen = {
		days = 730
	}

    # Here it comes
    option = {
		name = pbf_revolution.1.A		
		every_neighbor_country = {
            limit = {
                is_revolutionary = no
                ai = no
                is_subject = no
                NOT = { alliance_with = ROOT }
            }
            form_coalition_against = ROOT
        }
        set_global_flag = first_coalition
		
	}
}

    option = {

        every_country = {
            limit = {
                is_revolutionary = no
                ai = no
                is_subject = no
                NOT = { alliance_with = ROOT }
                OR = {
                    any_subject_country = {
                        is_neighbor_of = ROOT
                    }
                    is_neighbor_of = ROOT
                }
            }

            form_coalition_against = ROOT
        }	
        set_global_flag = second_coalition
		
	}
}
